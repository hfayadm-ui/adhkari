    35→/* =============================================
    36→   Helper: darken/lighten hex color
    37→   ============================================= */
    38→function adjustColor(hex: string, amount: number): string {
    39→  const num = parseInt(hex.replace('#', ''), 16);
    40→  const r = Math.min(255, Math.max(0, ((num >> 16) & 0xff) + amount));
    41→  const g = Math.min(255, Math.max(0, ((num >> 8) & 0xff) + amount));
    42→  const b = Math.min(255, Math.max(0, (num & 0xff) + amount));
    43→  return `rgb(${r},${g},${b})`;
    44→}

   194→          <g key={i}>
   195→            {/* Leaf shape */}
   196→            <path
   197→              d={`M${s/2+curve*0.6} ${topY} Q${cp1x} ${cp1y} ${tipX} ${tipY} Q${cp2x} ${cp2y} ${s/2+curve*0.6} ${topY}`}

   357→          <g key={i}>
   358→            <circle cx={cx} cy={cy} r={s*0.014} fill={i % 2 === 0 ? cfg.fruitColor : cfg.flowerColor} opacity='0.85' />
   359→            <circle cx={cx-0.5} cy={cy-0.5} r={s*0.005} fill='white' opacity='0.25' />
   360→          </g>
   361→        );
   362→      })}
   363→    </svg>
   364→  );
   365→}

   367→/* =============================================
   368→   Willow Tree — drooping branches
   369→   ============================================= */
   370→function WillowTree({ cfg, growth, s }: { cfg: TreeStage; growth: number; s: number }) {
   371→  const trunkH = s * cfg.trunk * Math.min(growth, 1);
   372→  const canopyR = s * cfg.canopy * Math.min(growth * 0.7, 0.7);
   373→  const canopyY = s * 0.92 - trunkH;
   374→  const branchCount = Math.min(Math.floor(growth * 8) + 1, 8);
   375→
   376→  return (
   377→    <svg viewBox={`0 0 ${s} ${s}`} className='w-full h-full'>
   378→      <defs>
   379→        <radialGradient id={`wl-${s}`} cx='50%' cy='30%'>
   380→          <stop offset='0%' stopColor={adjustColor(cfg.color, 25)} />
   381→          <stop offset='100%' stopColor={adjustColor(cfg.color, -15)} />
   382→        </radialGradient>
   383→      </defs>
   384→
   385→      <Ground s={s} color={cfg.color} growth={growth} />
   386→
   387→      {/* Trunk */}
   388→      <path d={`M${s/2-3} ${s*0.92} Q${s/2-4} ${s*0.92-trunkH*0.5} ${s/2-2} ${canopyY+5} Q${s/2} ${canopyY} ${s/2+2} ${canopyY+5} Q${s/2+4} ${s*0.92-trunkH*0.5} ${s/2+3} ${s*0.92}`} fill={cfg.trunkColor} />
   389→
   390→      {/* Canopy dome */}
   391→      <circle cx={s/2} cy={canopyY - canopyR * 0.2} r={canopyR * 0.6} fill={`url(#wl-${s})`} opacity='0.85' />
   392→      <circle cx={s/2 - canopyR * 0.2} cy={canopyY - canopyR * 0.1} r={canopyR * 0.5} fill={cfg.color} opacity='0.8' />
   393→      <circle cx={s/2 + canopyR * 0.2} cy={canopyY - canopyR * 0.1} r={canopyR * 0.5} fill={cfg.color} opacity='0.8' />
   394→
   395→      {/* Drooping willow branches */}
   396→      {branchCount > 2 && [...Array(branchCount)].map((_, i) => {
   397→        const angle = (i / branchCount) * Math.PI - Math.PI * 0.05;
   398→        const startX = s/2 + Math.cos(angle) * canopyR * 0.4;
   399→        const startY = canopyY - canopyR * 0.1;
   400→        const endX = startX + Math.cos(angle) * canopyR * 0.3 + (i % 2 ? 3 : -3);
   401→        const endY = s * 0.88;
   402→        const cpX = startX + (endX - startX) * 0.3 + (i % 2 ? 5 : -5);
   403→        const cpY = startY + (endY - startY) * 0.2;
   404→        return (
   405→          <path key={i}
   406→            d={`M${startX} ${startY} Q${cpX} ${cpY} ${endX} ${endY}`}
   407→            stroke={cfg.color} strokeWidth='1.2' fill='none' opacity={0.4 + (i % 3) * 0.1}
   408→            strokeLinecap='round'
   409→          />
   410→        );
   411→      })}
   412→
   413→      {/* Small leaves on branches */}
   414→      {growth >= 0.4 && [...Array(Math.min(Math.floor(growth * 12), 12))].map((_, i) => {
   415→        const t = (i + 1) / 13;
   416→        const angle = (i / 12) * Math.PI;
   417→        const bx = s/2 + Math.cos(angle) * canopyR * (0.3 + t * 0.3);
   418→        const by = canopyY + t * (s * 0.88 - canopyY);
   419→        return <circle key={i} cx={bx} cy={by} r={s * 0.008} fill={cfg.flowerColor} opacity='0.5' />;
   420→      })}
   421→    </svg>
   422→  );
   423→}

   425→/* =============================================
   426→   Magical Orbiting Lights — firefly-like particles
   427→   ============================================= */
   428→function OrbitingLights({ size, color, growth, cfg, count = 6 }: {
   429→  size: number; color: string; growth: number; cfg: TreeStage; count?: number;
   430→}) {
   431→  if (growth < 0.15) return null;
   432→
   433→  const actualCount = Math.min(Math.floor(growth * count) + 1, count);
   434→  const baseRadius = size * 0.4;
   435→
   436→  /* Generate deterministic but varied light configs */
   437→  const lights = useMemo(() => {
   438→    return [...Array(actualCount)].map((_, i) => {
   439→      const colors = [color, cfg.lightColor2, cfg.lightColor3, '#fbbf24', '#ffffff'];
   440→      return {
   441→        duration: 5 + (i * 2.3) % 6,
   442→        radiusX: baseRadius * (0.75 + (i % 4) * 0.12),
   443→        radiusY: baseRadius * (0.45 + (i % 3) * 0.1),
   444→        startAngle: (i / actualCount) * Math.PI * 2,
   445→        particleSize: 2.5 + (i % 3) * 1.5,
   446→        color: colors[i % colors.length],
   447→        direction: i % 3 === 0 ? -1 : 1,
   448→        wobble: 0.1 + (i % 4) * 0.05,
   449→        pulseSpeed: 1.5 + (i % 3) * 0.8,
   450→      };
   451→    });
   452→  }, [actualCount, baseRadius, color, cfg.lightColor2, cfg.lightColor3]);
   453→
   454→  return (
   455→    <>
   456→      {lights.map((light, i) => (
   457→        <motion.div
   458→          key={i}
   459→          className='absolute rounded-full pointer-events-none'
   460→          style={{
   461→            width: light.particleSize,
   462→            height: light.particleSize,
   463→            background: `radial-gradient(circle, white 0%, ${light.color} 40%, transparent 70%)`,
   464→            boxShadow: `0 0 ${light.particleSize * 2}px ${light.color}, 0 0 ${light.particleSize * 5}px ${light.color}60, 0 0 ${light.particleSize * 8}px ${light.color}25`,
   465→          }}
   466→          animate={{
   467→            x: light.direction === 1
   468→              ? [
   469→                  Math.cos(light.startAngle) * light.radiusX,
   470→                  Math.cos(light.startAngle + Math.PI * 0.5) * light.radiusX,
   471→                  Math.cos(light.startAngle + Math.PI) * light.radiusX,
   472→                  Math.cos(light.startAngle + Math.PI * 1.5) * light.radiusX,
   473→                  Math.cos(light.startAngle + Math.PI * 2) * light.radiusX,
   474→                ]
   475→              : [
   476→                  Math.cos(light.startAngle) * light.radiusX,
   477→                  Math.cos(light.startAngle - Math.PI * 0.5) * light.radiusX,
   478→                  Math.cos(light.startAngle - Math.PI) * light.radiusX,
   479→                  Math.cos(light.startAngle - Math.PI * 1.5) * light.radiusX,
   480→                  Math.cos(light.startAngle - Math.PI * 2) * light.radiusX,
   481→                ],
   482→            y: [
   483→              Math.sin(light.startAngle) * light.radiusY,
   484→              Math.sin(light.startAngle + Math.PI * 0.5 * light.direction) * light.radiusY * (1 + light.wobble),
   485→              Math.sin(light.startAngle + Math.PI) * light.radiusY,
   486→              Math.sin(light.startAngle + Math.PI * 1.5 * light.direction) * light.radiusY * (1 - light.wobble),
   487→              Math.sin(light.startAngle + Math.PI * 2) * light.radiusY,
   488→            ],
   489→            opacity: [0.3, 0.9, 0.5, 1, 0.3],
   490→            scale: [0.6, 1.3, 0.8, 1.15, 0.6],
   491→          }}
   492→          transition={{
   493→            duration: light.duration,
   494→            repeat: Infinity,
   495→            ease: 'easeInOut',
   496→            delay: i * 0.7,
   497→          }}
   498→        />
   499→      ))}
   500→    </>
   501→  );
   502→}

   504→/* =============================================
   505→   Floating Sparkle Particles
   506→   ============================================= */
   507→function FloatingParticles({ size, color, growth }: { size: number; color: string; growth: number }) {
   508→  if (growth < 0.3) return null;
   509→
   510→  const particles = useMemo(() => {
   511→    return [...Array(8)].map((_, i) => ({
   512→      x: size * (0.15 + (i * 0.1) % 0.7),
   513→      y: size * (0.1 + (i * 0.11) % 0.7),
   514→      size: 1 + (i % 3),
   515→      delay: i * 0.8,
   516→      duration: 3 + (i % 4),
   517→      drift: 5 + (i % 3) * 3,
   518→    }));
   519→  }, [size]);
   520→
   521→  return (
   522→    <>
   523→      {particles.map((p, i) => (
   524→        <motion.div
   525→          key={i}
   526→          className='absolute rounded-full pointer-events-none'
   527→          style={{
   528→            width: p.size,
   529→            height: p.size,
   530→            background: color,
   531→            boxShadow: `0 0 ${p.size * 2}px ${color}80`,
   532→          }}
   533→          animate={{
   534→            y: [p.y, p.y - p.drift, p.y - p.drift * 0.5, p.y],
   535→            x: [p.x, p.x + p.drift * 0.5, p.x - p.drift * 0.3, p.x],
   536→            opacity: [0, 0.7, 0.4, 0],
   537→            scale: [0, 1, 0.8, 0],
   538→          }}
   539→          transition={{
   540→            duration: p.duration,
   541→            repeat: Infinity,
   542→            ease: 'easeInOut',
   543→            delay: p.delay,
   544→          }}
   545→        />
   546→      ))}
   547→    </>
   548→  );
   549→}

   551→/* =============================================
   552→   Main Tree Visualization Component
   553→   ============================================= */
   554→export default function TreeVisualization({ treeType, level, size = 200 }: TreeVizProps) {
   555→  const cfg = treeConfigs[treeType] || treeConfigs.olive;
   556→  const growth = Math.min(level / 10, 1);
   557→  const s = size;
   558→
   559→  /* Deterministic sway based on treeType */
   560→  const swayAmount = growth > 0.1 ? (1.5 + growth * 2) : 0;
   561→  const swayDuration = 3.5 + (treeType === 'palm' ? 1.5 : treeType === 'willow' ? 1 : 0);
   562→  const lightCount = level >= 7 ? 10 : level >= 4 ? 7 : level >= 2 ? 5 : 3;
   563→
   564→  /* Render tree SVG based on type */
   565→  const treeSVG = useMemo(() => {
   566→    const props = { cfg, growth, s };
   567→    switch (treeType) {
   568→      case 'palm': return <PalmTree {...props} />;
   569→      case 'olive': return <OliveTree {...props} />;
   570→      case 'pomegranate': return <PomegranateTree {...props} />;
   571→      case 'cedar': return <CedarTree {...props} />;
   572→      case 'sidrah': return <SidrahTree {...props} />;
   573→      case 'willow': return <WillowTree {...props} />;
   574→      default: return <OliveTree {...props} />;
   575→    }
   576→  }, [treeType, cfg, growth, s]);
   577→
   578→  return (
   579→    <motion.div
   580→      initial={{ scale: 0.8, opacity: 0 }}
   581→      animate={{ scale: 1, opacity: 1 }}
   582→      transition={{ type: 'spring', stiffness: 200, damping: 15 }}
   583→      style={{ width: size, height: size }}
   584→      className='relative flex items-center justify-center'
   585→    >
   586→      {/* Outer atmospheric glow */}
   587→      <motion.div
   588→        className='absolute rounded-full pointer-events-none'
   589→        style={{
   590→          inset: -size * 0.18,
   591→          background: `radial-gradient(ellipse at 50% 45%, ${cfg.color}18, ${cfg.color}08 40%, transparent 65%)`,
   592→        }}
   593→        animate={growth > 0.2 ? {
   594→          scale: [1, 1.12, 0.95, 1.08, 1],
   595→          opacity: [0.4, 0.7, 0.5, 0.65, 0.4],
   596→        } : {}}
   597→        transition={{ duration: 4, repeat: Infinity, ease: 'easeInOut' }}
   598→      />
   599→
   600→      {/* Second glow layer — colored aura */}
   601→      {growth > 0.4 && (
   602→        <motion.div
   603→          className='absolute rounded-full pointer-events-none'
   604→          style={{
   605→            inset: -size * 0.08,
   606→            background: `radial-gradient(ellipse at 50% 40%, ${cfg.lightColor2}12, transparent 60%)`,
   607→          }}
   608→          animate={{
   609→            scale: [1, 1.08, 1],
   610→            opacity: [0.3, 0.6, 0.3],
   611→          }}
   612→          transition={{ duration: 3, repeat: Infinity, ease: 'easeInOut', delay: 0.5 }}
   613→        />
   614→      )}
   615→
   616→      {/* Swaying tree container — organic movement */}
   617→      <motion.div
   618→        className='relative'
   619→        style={{ width: s, height: s, transformOrigin: '50% 95%' }}
   620→        animate={swayAmount > 0 ? {
   621→          rotate: [0, swayAmount, -swayAmount * 0.3, -swayAmount, swayAmount * 0.3, 0],
   622→        } : {}}
   623→        transition={{
   624→          duration: swayDuration,
   625→          repeat: Infinity,
   626→          ease: [0.45, 0.05, 0.55, 0.95],
   627→        }}
   628→      >
   629→        {/* Tree SVG */}
   630→        {treeSVG}
   631→
   632→        {/* Inner canopy glow — breathing */}
   633→        {growth > 0.3 && (
   634→          <motion.div
   635→            className='absolute pointer-events-none rounded-full'
   636→            style={{
   637→              width: s * 0.45,
   638→              height: s * 0.4,
   639→              top: s * 0.08,
   640→              left: s * 0.28,
   641→              background: `radial-gradient(circle, ${cfg.color}25, transparent 65%)`,
   642→            }}
   643→            animate={{
   644→              opacity: [0.2, 0.6, 0.3, 0.55, 0.2],
   645→              scale: [0.95, 1.05, 1, 1.03, 0.95],
   646→            }}
   647→            transition={{ duration: 3.5, repeat: Infinity, ease: 'easeInOut' }}
   648→          />
   649→        )}
   650→      </motion.div>
   651→
   652→      {/* Orbiting magical lights */}
   653→      <OrbitingLights size={s} color={cfg.color} growth={growth} cfg={cfg} count={lightCount} />
   654→
   655→      {/* Floating sparkle particles */}
   656→      <FloatingParticles size={s} color={cfg.color} growth={growth} />
   657→    </motion.div>
   658→  );
   659→}
