   180→// ==================== مكون الرسوم المتحركة للتنفس ====================
   181→function BreathingCircle({ phase, isActive, totalTime, elapsed }: {
   182→  phase: string;
   183→  isActive: boolean;
   184→  totalTime: number;
   185→  elapsed: number;
   186→}) {
   187→  const isInhale = phase === 'شهيق';
   188→  const isHold = phase === 'حبس';
   189→  const isExhale = phase === 'زفير';
   190→
   191→  const scaleValue = isActive
   192→    ? isInhale ? 1.35 : isExhale ? 0.85 : 1.15
   193→    : 1;
   194→
   195→  const glowColor = isInhale
   196→    ? 'rgba(59, 130, 246, 0.4)'
   197→    : isExhale
   198→    ? 'rgba(147, 51, 234, 0.4)'
   199→    : 'rgba(16, 185, 129, 0.3)';
   200→
   201→  const progress = totalTime > 0 ? (elapsed % totalTime) / totalTime : 0;
   202→
   203→  return (
   204→    <div className='relative flex items-center justify-center my-6'>
   205→      {/* Outer glow rings */}
   206→      <motion.div
   207→        className='absolute w-52 h-52 rounded-full'
   208→        animate={{
   209→          scale: isActive ? [1, scaleValue * 1.15] : 1,
   210→          opacity: isActive ? [0.15, 0.3, 0.15] : 0.1,
   211→        }}
   212→        transition={{ duration: totalTime || 4, repeat: isActive ? Infinity : 0, ease: 'easeInOut' }}
   213→        style={{ background: `radial-gradient(circle, ${glowColor}, transparent)` }}
   214→      />
   215→      <motion.div
   216→        className='absolute w-40 h-40 rounded-full'
   217→        animate={{
   218→          scale: isActive ? [1, scaleValue * 1.1] : 1,
   219→          opacity: isActive ? [0.2, 0.4, 0.2] : 0.15,
   220→        }}
   221→        transition={{ duration: totalTime || 4, repeat: isActive ? Infinity : 0, ease: 'easeInOut' }}
   222→        style={{ background: `radial-gradient(circle, ${glowColor}, transparent)` }}
   223→      />
   224→
   225→      {/* Main circle */}
   226→      <motion.div
   227→        className='relative w-36 h-36 rounded-full flex flex-col items-center justify-center'
   228→        style={{
   229→          background: 'var(--app-surface)',
   230→          border: '2px solid var(--app-border)',
   231→          boxShadow: isActive ? `0 0 40px ${glowColor}, 0 0 80px ${glowColor}` : '0 0 20px rgba(212,175,55,0.1)',
   232→        }}
   233→        animate={{
   234→          scale: isActive ? [1, scaleValue] : 1,
   235→        }}
   236→        transition={{ duration: totalTime || 4, repeat: isActive ? Infinity : 0, ease: 'easeInOut' }}
   237→      >
   238→        <motion.span
   239→          className='text-2xl font-bold app-text'
   240→          animate={{ opacity: isActive ? 1 : 0.5 }}
   241→        >
   242→          {phase || '...' }
   243→        </motion.span>
   244→        {isActive && (
   245→          <motion.span
   246→            className='text-xs app-text-2 mt-1'
   247→            initial={{ opacity: 0 }}
   248→            animate={{ opacity: 1 }}
   249→          >
   250→            {Math.ceil(totalTime - (elapsed % totalTime))}ث
   251→          </motion.span>
   252→        )}
   253→      </motion.div>
   254→
   255→      {/* Orbiting particles */}
   256→      {isActive && [0, 1, 2].map((i) => (
   257→        <motion.div
   258→          key={i}
   259→          className='absolute w-2 h-2 rounded-full bg-amber-400/60'

   261→            rotate: [0 + i * 120, 360 + i * 120],

   263→          transition={{ duration: 12, repeat: Infinity, ease: 'linear' }}
   264→          style={{
   265→            transformOrigin: '120px center',
   266→          }}
   267→        />
   268→      ))}
   269→    </div>

   273→// ==================== مكون حلقة التنفس ====================
   274→function BreathingSession({ technique, onBack }: { technique: BreathingTechnique; onBack: () => void }) {
   275→  const [isActive, setIsActive] = useState(false);
   276→  const [currentCycle, setCurrentCycle] = useState(0);
   277→  const [currentPhaseIndex, setCurrentPhaseIndex] = useState(0);
   278→  const [elapsed, setElapsed] = useState(0);
   279→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   280→  const startTimeRef = useRef<number>(0);
   281→  const pauseTimeRef = useRef<number>(0);
   282→
   283→  const currentPhase = technique.phases[currentPhaseIndex];
   284→  const totalDuration = technique.phases.reduce((s, p) => s + p.duration, 0);
   285→  const totalSessionTime = totalDuration * technique.cycles;
   286→  const isFinished = currentCycle >= technique.cycles;
   287→
   288→  const getCurrentPhaseElapsed = useCallback(() => {
   289→    const elapsedInCycles = elapsed % totalDuration;
   290→    let acc = 0;
   291→    for (let i = 0; i < technique.phases.length; i++) {
   292→      if (elapsedInCycles < acc + technique.phases[i].duration) {
   293→        setCurrentPhaseIndex(i);
   294→        return elapsedInCycles - acc;
   295→      }
   296→      acc += technique.phases[i].duration;
   297→    }
   298→    return 0;
   299→  }, [elapsed, totalDuration, technique.phases]);
   300→
   301→  useEffect(() => {
   302→    if (isActive) {
   303→      startTimeRef.current = Date.now() - pauseTimeRef.current;
   304→      timerRef.current = setInterval(() => {
   305→        const e = Math.floor((Date.now() - startTimeRef.current) / 1000);
   306→        setElapsed(e);
   307→        const cycle = Math.floor(e / totalDuration);
   308→        setCurrentCycle(cycle);
   309→        if (cycle >= technique.cycles) {
   310→          setIsActive(false);
   311→          if (timerRef.current) clearInterval(timerRef.current);
   312→        }
   313→      }, 100);
   314→    } else {
   315→      if (timerRef.current) clearInterval(timerRef.current);
   316→      pauseTimeRef.current = elapsed * 1000;
   317→    }
   318→    return () => { if (timerRef.current) clearInterval(timerRef.current); };
   319→  }, [isActive, totalDuration, technique.cycles, elapsed]);
   320→
   321→  useEffect(() => {
   322→    getCurrentPhaseElapsed();
   323→  }, [getCurrentPhaseElapsed]);
   324→
   325→  const reset = () => {
   326→    setIsActive(false);
   327→    setCurrentCycle(0);
   328→    setCurrentPhaseIndex(0);
   329→    setElapsed(0);
   330→    pauseTimeRef.current = 0;
   331→  };
   332→
   333→  const overallProgress = Math.min(elapsed / totalSessionTime, 1);
   334→
   335→  return (
   336→    <motion.div
   337→      initial={{ opacity: 0, y: 20 }}
   338→      animate={{ opacity: 1, y: 0 }}
   339→      exit={{ opacity: 0, y: -20 }}
   340→      className='min-h-screen flex flex-col'
   341→    >
   342→      {/* Header */}
   343→      <div className='flex items-center justify-between px-4 pt-4 pb-2'>
   344→        <button onClick={onBack} className='w-10 h-10 rounded-full app-surface flex items-center justify-center app-surface-h'>
   345→          <ChevronLeft className='w-5 h-5 app-text' />
   346→        </button>
   347→        <h2 className='app-text font-bold text-base'>{technique.name}</h2>
   348→        <button onClick={reset} className='w-10 h-10 rounded-full app-surface flex items-center justify-center app-surface-h'>
   349→          <RotateCcw className='w-4 h-4 app-text-2' />
   350→        </button>
   351→      </div>
   352→
   353→      {/* Progress bar */}
   354→      <div className='mx-4 mb-2'>
   355→        <div className='h-1.5 rounded-full app-surface overflow-hidden'>
   356→          <motion.div
   357→            className='h-full rounded-full bg-gradient-to-l from-amber-400 to-amber-500'
   358→            style={{ width: `${overallProgress * 100}%` }}
   359→          />
   360→        </div>
   361→        <div className='flex justify-between mt-1'>
   362→          <span className='app-text-muted text-[10px]'>الدورة {Math.min(currentCycle + 1, technique.cycles)}/{technique.cycles}</span>
   363→          <span className='app-text-muted text-[10px]'>{Math.floor(elapsed / 60)}:{String(elapsed % 60).padStart(2, '0')}</span>
   364→        </div>
   365→      </div>
   366→
   367→      {/* Breathing animation */}
   368→      <div className='flex-1 flex flex-col items-center justify-center px-4'>
   369→        <BreathingCircle
   370→          phase={isFinished ? 'تم!' : currentPhase?.name || '...'}
   371→          isActive={isActive}
   372→          totalTime={currentPhase?.duration || 4}
   373→          elapsed={elapsed}
   374→        />
   375→        {/* Instruction */}
   376→        <AnimatePresence mode='wait'>
   377→          <motion.p
   378→            key={currentPhaseIndex}
   379→            initial={{ opacity: 0, y: 10 }}
   380→            animate={{ opacity: 1, y: 0 }}
   381→            exit={{ opacity: 0, y: -10 }}
   382→            className='app-text-2 text-sm text-center mt-4 mb-2'
   383→          >
   384→            {isFinished ? 'أحسنت! أكملت جميع الدورات' : currentPhase?.instruction || ''}
   385→          </motion.p>
   386→        </AnimatePresence>
   387→
   388→        {/* Phase indicators */}
   389→        <div className='flex items-center gap-2 mt-3'>
   390→          {technique.phases.map((p, i) => (
   391→            <div
   392→              key={i}
   393→              className={`px-3 py-1.5 rounded-full text-xs transition-all duration-300 ${
   394→                i === currentPhaseIndex && isActive
   395→                  ? 'bg-amber-500/20 text-amber-300 border border-amber-500/30'
   396→                  : 'app-surface app-text-muted'
   397→              }`}
   398→            >
   399→              {p.name}
   400→            </div>
   401→          ))}
   402→        </div>
   403→
   404→        {/* Islamic note */}
   405→        {technique.islamicNote && (
   406→          <motion.div
   407→            initial={{ opacity: 0 }}
   408→            animate={{ opacity: 1 }}
   409→            transition={{ delay: 0.5 }}
   410→            className='mt-6 text-center max-w-xs'
   411→          >
   412→            <p className='text-amber-400/60 text-[11px] leading-relaxed'>{technique.islamicNote}</p>
   413→          </motion.div>
   414→        )}
   415→      </div>
   416→
   417→      {/* Control button */}
   418→      <div className='px-4 pb-8 pt-4 flex justify-center'>
   419→        <motion.button
   420→          whileTap={{ scale: 0.95 }}
   421→          onClick={() => !isFinished && setIsActive(!isActive)}
   422→          className={`w-16 h-16 rounded-full flex items-center justify-center shadow-lg transition-all ${
   423→            isFinished
   424→              ? 'bg-emerald-500/20 border border-emerald-500/30'
   425→              : isActive
   426→              ? 'bg-amber-500/20 border border-amber-500/30'
   427→              : 'bg-gradient-to-br from-amber-500 to-amber-600 shadow-amber-500/20'
   428→          }`}
   429→        >
   430→          {isFinished ? (
   431→            <IslamicIcon name='check-circle' className='w-7 h-7 text-emerald-400' />

   444→function RelaxationToolViewer({ tool, onBack }: { tool: RelaxationTool; onBack: () => void }) {
   445→  const [currentStep, setCurrentStep] = useState(0);
   446→  const [isComplete, setIsComplete] = useState(false);

   453→    }
   454→  };

   457→    setCurrentStep(0);
   458→    setIsComplete(false);

   462→
   463→  return (

   478→        </button>
   479→      </div>

   486→            style={{ width: `${progress * 100}%` }}
   487→            transition={{ duration: 0.3 }}
   488→          />
   489→        </div>
   490→        <span className='app-text-muted text-[10px] mt-1 block'>{currentStep + 1} / {tool.content.length}</span>
   491→      </div>
   492→
   493→      <div className='flex-1 flex flex-col items-center justify-center px-6'>
   494→        <AnimatePresence mode='wait'>
   495→          {isComplete ? (
   496→            <motion.div
   497→              key='complete'
   498→              initial={{ opacity: 0, scale: 0.8 }}
   499→              animate={{ opacity: 1, scale: 1 }}
   500→              className='text-center'

   502→              <motion.div
   503→                initial={{ scale: 0 }}
   504→                animate={{ scale: 1 }}
   505→                transition={{ type: 'spring', stiffness: 200 }}
   506→                className='w-20 h-20 mx-auto mb-6 rounded-full bg-emerald-500/20 border border-emerald-500/30 flex items-center justify-center'
   507→              >
   508→                <IslamicIcon name='check-circle' className='w-10 h-10 text-emerald-400' />
   509→              </motion.div>
   510→              <h3 className='app-text text-xl font-bold mb-2'>أحسنت!</h3>
   511→              <p className='app-text-2 text-sm mb-6'>أكملت الجلسة بنجاح، نسأل الله أن يريح قلبك</p>
   512→              <motion.button
   513→                whileTap={{ scale: 0.95 }}
   514→                onClick={reset}
   515→                className='px-6 py-3 rounded-xl bg-gradient-to-r from-amber-500 to-amber-600 text-white font-medium text-sm shadow-lg shadow-amber-500/20'
   516→              >
   517→                إعادة الجلسة
   518→              </motion.button>
   519→            </motion.div>
   520→          ) : (
   521→            <motion.div
   522→              key={currentStep}
   523→              initial={{ opacity: 0, x: 30 }}
   524→              animate={{ opacity: 1, x: 0 }}
   525→              exit={{ opacity: 0, x: -30 }}
   526→              className='text-center w-full'
   527→            >
   528→              {/* Step number */}
   529→              <div className='w-12 h-12 mx-auto mb-6 rounded-full bg-gradient-to-br from-amber-500/20 to-amber-600/10 border border-amber-500/20 flex items-center justify-center'>
   530→                <span className='text-amber-400 font-bold text-lg'>{currentStep + 1}</span>
   531→              </div>
   532→
   533→              {/* Step text */}
   534→              <p className='app-text text-lg leading-loose mb-8' style={{ fontFamily: 'var(--font-arabic)' }}>
   535→                {tool.content[currentStep]}
   536→              </p>
   537→
   538→              {/* Islamic context for first step */}
   539→              {currentStep === 0 && tool.islamicContext && (
   540→                <motion.p
   541→                  initial={{ opacity: 0 }}
   542→                  animate={{ opacity: 1 }}
   543→                  transition={{ delay: 0.3 }}
   544→                  className='text-amber-400/50 text-xs leading-relaxed mb-6'

   549→            </motion.div>
   550→          )}
   551→        </AnimatePresence>
   552→      </div>
   553→
   554→      {/* Next button */}
   555→      {!isComplete && (
   556→        <div className='px-4 pb-8 flex justify-center'>
   557→          <motion.button
   558→            whileTap={{ scale: 0.95 }}
   559→            onClick={next}
   560→            className='w-full max-w-xs py-3.5 rounded-xl bg-gradient-to-r from-amber-500 to-amber-600 text-white font-bold text-sm shadow-lg shadow-amber-500/20'
   561→          >
   562→            {currentStep < tool.content.length - 1 ? 'الخطوة التالية' : 'إنهاء الجلسة'}
   563→          </motion.button>
   564→        </div>
   565→      )}
   566→    </motion.div>
   567→  );
   568→}
