   340→function GardenPlantSVG({ type, size, color, delay = 0 }: { type: string; size: number; color: string; delay?: number }) {

   354→      <PlantComponent size={size} color={color} />

   356→      {[...Array(3)].map((_, i) => (

   358→          key={i}
   359→          className='absolute w-1 h-1 rounded-full'
